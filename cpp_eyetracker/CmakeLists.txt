# /CmakeLists.txt

# cmake minimum version
cmake_minimum_required(VERSION 3.15)

if(NOT DEFINED ENV{VCPKG_ROOT})
    message(WARNING "VCPKG_ROOT not set, please specify path to vcpkg, e.g.:")
    message(WARNING "cmake -D VCPKG_ROOT=C:/vcpkg [...]")
endif()

if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    message(FATAL_ERROR
        "CMAKE_TOOLCHAIN_FILE is not set.\n"
        "Please configure with:\n"
        "  -DCMAKE_TOOLCHAIN_FILE=D:/apps/vcpkg/scripts/buildsystems/vcpkg.cmake"
    )
endif()

# project name (https://cmake.com.cn/cmake/help/latest/command/project.html)
# ⚠️ do not set more than one project under the premier directory,
#    unless you want to create anoter project in the same directory.
project(cpp_eyetracker LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# set(CMAKE_VERBOSE_MAKEFILE ON)

add_compile_options(/utf-8)

# choose build type
option(BUILD_TESTING "Build tests" ON)
if(BUILD_TESTING)
    enable_testing()
    add_subdirectory(tests)
endif()

# add subdirectories
add_subdirectory(cfg)
add_subdirectory(core)
add_subdirectory(utils)
add_subdirectory(calib)
add_subdirectory(inference)
