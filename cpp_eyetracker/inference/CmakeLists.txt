# /inference/CmakeLists.txt

# add static library for inference
add_library(inference STATIC
    src/one_camera_spherical.cpp
)
add_library(eyetracker::inference ALIAS inference)

target_include_directories(inference PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)

find_package(Ceres CONFIG REQUIRED)

target_link_libraries(inference 
    PUBLIC
        eyetracker::utils
        Ceres::ceres
)
