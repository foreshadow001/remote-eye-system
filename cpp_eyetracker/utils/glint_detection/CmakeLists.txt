# /utils/glint_detection/CmakeLists.txt

find_package(OpenCV CONFIG REQUIRED)

add_library(glint_detection STATIC
    src/detect_glint.cpp
)

add_library(eyetracker::glint_detection ALIAS glint_detection)

target_include_directories(glint_detection
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
        $<INSTALL_INTERFACE:include>
)

target_link_libraries(glint_detection
    PUBLIC
        eyetracker::cfg
        eyetracker::utils
        ${OpenCV_LIBS}
)
