add_subdirectory(cam_model)

    src/gaze_estimation_types.cpp
    src/intersection.cpp
    src/visualize.cpp
)

)

target_link_libraries(utils PUBLIC
    eyetracker::cfg
    eyetracker::cam_model
    ${OpenCV_LIBS}
)
