# cam/CMakeLists.txt
cmake_minimum_required(VERSION 3.14)

add_library(cam STATIC
    src/basler.cpp
)
add_library(eyetracker::cam ALIAS cam)

# 包含头文件
target_include_directories(cam PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
)

if(NOT DEFINED ENV{PYLON_DEV_DIR})
    message(WARNING "⚠️  PYLON_DEV_DIR not set. Please specify path to Pylon SDK, e.g.:")
    message(WARNING "    cmake -D PYLON_DEV_DIR=\"C:/Program Files/Basler/pylon 7\" ..")
endif()

set(PYLON_DEV_DIR $ENV{PYLON_DEV_DIR})

if(PYLON_DEV_DIR)
    target_include_directories(cam PUBLIC
        "${PYLON_DEV_DIR}/include"
    )

    if(WIN32)
        if(EXISTS "${PYLON_DEV_DIR}/lib/x64")
            target_link_directories(cam PUBLIC "${PYLON_DEV_DIR}/lib/x64")
        elseif(EXISTS "${PYLON_DEV_DIR}/lib")
            target_link_directories(cam PUBLIC "${PYLON_DEV_DIR}/lib")
        endif()
    else()
        target_link_directories(cam PUBLIC "${PYLON_DEV_DIR}/lib64")
    endif()
endif()

find_package(OpenCV REQUIRED)

target_link_libraries(cam PUBLIC
    ${OpenCV_LIBS}
    PylonBase_v10
    GCBase_MD_VC141_v3_1_Basler_pylon_v3.lib
    GenApi_MD_VC141_v3_1_Basler_pylon_v3.lib
    PylonUtility_v10.lib
)
