find_package(OpenCV REQUIRED)

cmake_minimum_required(VERSION 3.14)

set(TEST_CAM_SOURCES
    test_cam.cpp
)

add_executable(test_cam ${TEST_CAM_SOURCES})

target_link_libraries(test_cam
    PRIVATE
        eyetracker::cfg
        eyetracker::cam
)

set_target_properties(test_cam PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)

message(STATUS "✅ test_cam will be built and linked with eyetracker::cam")
